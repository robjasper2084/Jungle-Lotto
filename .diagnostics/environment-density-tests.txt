
import {LEAF_PATCH_SCALE,thickenLeafIslands} from './vegetationVolume.ts';
import {hedgePixels,createHedgeTexture} from './mazeHedgeTexture.ts';
test('individual leaf quads get denser while their centres, topology and source stay fixed',()=>{const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute([-1,-1,0,1,-1,0,1,1,0,-1,1,0,9,-1,0,11,-1,0,11,1,0,9,1,0],3));g.setIndex([0,1,2,0,2,3,4,5,6,4,6,7]);const source=g.clone(),indices=[...g.index!.array];assert.equal(thickenLeafIslands(g),2);assert.deepEqual([...g.index!.array],indices);for(const offset of [0,4]){const x=Array.from({length:4},(_,i)=>g.attributes.position.getX(i+offset)),y=Array.from({length:4},(_,i)=>g.attributes.position.getY(i+offset));assert(Math.abs(x.reduce((a,b)=>a+b,0)/4-(offset?10:0))<1e-6);assert(Math.abs(Math.max(...x)-Math.min(...x)-2*LEAF_PATCH_SCALE)<1e-6);assert(Math.abs(y.reduce((a,b)=>a+b,0))<1e-6);}assert.equal(source.attributes.position.getX(0),-1);g.dispose();source.dispose();});
test('connected large geometry and non-indexed shapes do not receive leaf-island scaling',()=>{const g=new T.PlaneGeometry(4,4,4,4),before=[...g.attributes.position.array];assert.equal(thickenLeafIslands(g),0);assert.deepEqual([...g.attributes.position.array],before);const flat=g.toNonIndexed();assert.equal(thickenLeafIslands(flat),0);g.dispose();flat.dispose();});
test('leaf clusters use deterministic opaque bounded texture data with visible variation',()=>{const a=hedgePixels(),b=hedgePixels();assert.equal(a.length,128*128*4);assert.deepEqual(a,b);let low=255,high=0;for(let i=0;i<a.length;i+=4){assert.equal(a[i+3],255);low=Math.min(low,a[i+1]);high=Math.max(high,a[i+1]);}assert(high-low>40);assert.throws(()=>hedgePixels(0),RangeError);assert.throws(()=>hedgePixels(257),RangeError);const texture=createHedgeTexture();assert.equal(texture.colorSpace,T.SRGBColorSpace);assert.equal(texture.wrapS,T.RepeatWrapping);assert.equal(texture.wrapT,T.RepeatWrapping);texture.dispose();});
