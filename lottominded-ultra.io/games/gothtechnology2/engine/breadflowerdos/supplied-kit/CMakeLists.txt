cmake_minimum_required(VERSION 3.20)
project(StaticBreadflowerInputBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(static_input_bridge STATIC engine/src/input_bridge.cpp)
target_include_directories(static_input_bridge PUBLIC engine/include PRIVATE third_party/breadflowerdos/src)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(static_input_bridge PRIVATE -Wall -Wextra -fno-fast-math)
endif()
add_executable(native_input_tests tests/native_test.cpp)
target_link_libraries(native_input_tests PRIVATE static_input_bridge)
enable_testing()
add_test(NAME native_input_tests COMMAND native_input_tests)
